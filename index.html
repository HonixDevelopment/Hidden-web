Calculate Student Grade

marks = int(input("Enter marks: "))
if marks > 90:
    grade = "A"
elif marks >= 70:
    grade = "B"
elif marks >= 50:
    grade = "C"
else:
    grade = "D"
print("Grade:", grade)

Multiplication Table using While Loop

n = int(input("Enter a number: "))
i = 1
while i <= 10:
    print(n, "x", i, "=", n*i)
    i += 1

Check Positive/Negative/Zero

num = int(input("Enter a number: "))
if num > 0:
    print("Positive")
elif num < 0:
    print("Negative")
else:
    print("Zero")

Print Odd Numbers 1-50

Mean, Median, Mode using NumPy

import numpy as np
from scipy import stats
data = [10,20,20,30,40,40,40,50]
print("Mean:", np.mean(data))
print("Median:", np.median(data))
print("Mode:", stats.mode(data)[0][0])
